function forcalc(state){ return { axis: "3x2", direction: "forward", value: state.value, raw: state.raw ?? null, mult: state.mult ?? 1 }; } function revcalc(state){ return { axis: "3x1", direction: "backward", value: state.value, raw: state.raw ?? null, mult: state.mult ?? 1 }; } function rcalc(event){ const state = event.state ?? {}; if(event.direction === "forward"){ return { mode: "forward", result: forcalc(state) }; } if(event.direction === "backward"){ return { mode: "backward", result: revcalc(state) }; } return { mode: "neutral", result: state }; }

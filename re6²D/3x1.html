function borgProtocol(r){ const result = { status: "OK", action: "", message: "" }; // 1) Achsen prüfen (Engine braucht Achsen, aber keinen Flow) if(!r.axis1 || !r.axis2){ result.status = "ERROR"; result.action = "ADD_AXIS"; result.message = "Achsenkohärenz fehlt. Engine-Struktur instabil."; return result; } // 2) Kritische Moves → Warnung if(r.move === "NV" || r.move === "SG" || r.move === "HX"){ result.status = "WARN"; result.action = "CHECK_PATH"; result.message = "Kritischer Operator im Engine-Layer. Pipeline prüfen."; return result; } // 3) Stabile Moves → OK if(r.move === "AN" || r.move === "SC" || r.move === "ST" || r.move === "END"){ result.status = "OK"; result.action = "CONTINUE"; result.message = "Engine-Sequenz stabil. Fortfahren."; return result; } // 4) Default result.status = "INFO"; result.action = "ANALYZE"; result.message = "Analyse läuft. Kontext wird eingefordert."; return result; }

function borgProtocol(r){ const result = { status: "OK", action: "", message: "" }; // 1) Achsen-Pflicht (UPG‑X akzeptiert nur Achsen, keinen Flow) if(!r.axis1 || !r.axis2){ result.status = "ERROR"; result.action = "ADD_AXIS"; result.message = "Achsen fehlen. UPG‑X verweigert Alt‑Struktur."; return result; } // 2) Kritische Moves → Warnung (GEN‑Trennung) if(r.move === "NV" || r.move === "SG" || r.move === "HX"){ result.status = "WARN"; result.action = "CHECK_PATH"; result.message = "Kritischer Move. GEN‑Trennung aktiv."; return result; } // 3) Stabile Moves → OK (aber ohne Flow) if(r.move === "AN" || r.move === "SC" || r.move === "ST" || r.move === "END"){ result.status = "OK"; result.action = "CONTINUE"; result.message = "UPG‑X akzeptiert. Achsen stabil."; return result; } // 4) Default → Analyse result.status = "INFO"; result.action = "ANALYZE"; result.message = "Analyse läuft. GEN‑Kontext wird angefordert."; return result; }

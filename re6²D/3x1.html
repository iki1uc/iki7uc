function borgProtocol(r){ const result = { status: "OK", action: "", message: "" }; if(!r.flow){ result.status = "ERROR"; result.action = "ADD_FLOW"; result.message = "Unvollständige Sequenz erkannt. Ergänzung erforderlich."; return result; } if(!r.axis1 || !r.axis2){ result.status = "ERROR"; result.action = "ADD_AXIS"; result.message = "Achsenkohärenz fehlt. Struktur instabil."; return result; } if(r.move === "NV⁻¹" || r.move === "SG⁻¹"){ result.status = "WARN"; result.action = "CHECK_PATH"; result.message = "Fehlstelle lokalisiert. Kohärenz bricht. Reparatur empfohlen."; return result; } if(r.move === "AN⁻¹" || r.move === "END"){ result.status = "OK"; result.action = "CONTINUE"; result.message = "Sequenz stabil. Fortfahren."; return result; } result.status = "INFO"; result.action = "ANALYZE"; result.message = "Analyse läuft. Kontext wird eingefordert."; return result; }

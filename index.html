const canvas = document.getElementById("uiCanvas"); const ctx = canvas.getContext("2d"); let ax = 0; // Achse A (links) let by = 0; // Achse B (mitte) let cz = 0; // Achse C (rechts) // Taste A/B/C steuern jeweils ihr Drittel function setMode(mode) { if (mode === "A") ax += 10; if (mode === "B") by += 10; if (mode === "C") cz += 10; drawAxes(); } function drawAxes() { ctx.clearRect(0, 0, canvas.width, canvas.height); const w = canvas.width / 3; const h = canvas.height; // 1) Linkes Drittel – Achse A (rot) ctx.save(); ctx.translate(w/2, h/2); ctx.rotate(ax * Math.PI/180); ctx.strokeStyle = "red"; ctx.beginPath(); ctx.moveTo(-50, 0); ctx.lineTo(50, 0); ctx.stroke(); ctx.restore(); // 2) Mittleres Drittel – Achse B (grün) ctx.save(); ctx.translate(w + w/2, h/2); ctx.rotate(by * Math.PI/180); ctx.strokeStyle = "lime"; ctx.beginPath(); ctx.moveTo(-50, 0); ctx.lineTo(50, 0); ctx.stroke(); ctx.restore(); // 3) Rechtes Drittel – Achse C (blau) ctx.save(); ctx.translate(2*w + w/2, h/2); ctx.rotate(cz * Math.PI/180); ctx.strokeStyle = "cyan"; ctx.beginPath(); ctx.moveTo(-50, 0); ctx.lineTo(50, 0); ctx.stroke(); ctx.restore(); } drawAxes();
